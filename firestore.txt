rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{uid} {
      allow read: if request.auth != null;
      allow create: if request.auth.uid == uid;
      allow update: if request.auth != null && (
        request.auth.uid == uid ||
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['friendIds', 'pendingPartyInvites'])
      );
    }

    match /friendRequests/{id} {
      allow read, write: if request.auth != null;
    }

    match /watchParties/{partyId} {
      allow read: if request.auth != null && (
        request.auth.uid in resource.data.memberIds ||
        request.auth.uid in resource.data.invitedIds
      );
      allow create: if request.auth != null;
      allow update: if request.auth != null && (
        request.auth.uid in resource.data.memberIds ||
        request.auth.uid in resource.data.invitedIds
      );

      match /ratings/{ratingId} {
        allow read, write: if request.auth != null;
      }
    }

    match /users/{userId}/marathons/{marathonId} {
      allow read, delete: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null && request.auth.uid == userId
        && request.resource.data.keys().hasOnly([
          'schemaVersion', 'title', 'universeId', 'order', 'entries', 'media', 'updatedAt'
        ])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 120
        && request.resource.data.order == 'custom'
        && request.resource.data.entries is list
        && request.resource.data.entries.size() > 0
        && request.resource.data.entries.size() <= 100
        && request.resource.data.media is map
        && request.resource.data.media.size() <= 100;
    }
    match /users/{userId}/marathonRuns/{runId} {
      allow read, delete: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['schemaVersion', 'completed', 'updatedAt'])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.completed is map;
    }

    match /users/{userId}/marathonPhotos/{photoId} {
      allow read, delete: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['schemaVersion', 'contentType', 'bytes', 'updatedAt'])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.contentType == 'image/jpeg'
        && request.resource.data.bytes is bytes
        && request.resource.data.bytes.size() > 0
        && request.resource.data.bytes.size() <= 184320;
    }

  }
}
